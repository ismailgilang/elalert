Simple API

From zero to dialog
in 3 lines.

typescript 
import { Dialog } from '@maellana25/elalert';
import '@maellana25/elalert/style.css';

// ✅ Simple alert
Dialog.success('Data saved successfully!');

// ✅ Async confirm with full type safety
const result = await Dialog.confirm({
  title: 'Delete record?',
  message: 'This action cannot be undone.',
  confirmText: 'Yes, delete',
});

if (result.isConfirmed) {
  await deleteRecord();
}

// ✅ Loading with live update
const loader = Dialog.loading({ message: 'Uploading...' });
await upload();
loader.close();
